add_executable(fiso-cli main.cpp)
set_target_properties(fiso-cli PROPERTIES OUTPUT_NAME fiso)
target_link_libraries(fiso-cli PRIVATE fiso fiso_flac)
if(WIN32)
  target_link_libraries(fiso-cli PRIVATE shell32)  # CommandLineToArgvW (UTF-8 arguments)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(fiso-cli PRIVATE -static-libstdc++ -static-libgcc)
endif()
install(TARGETS fiso-cli)
