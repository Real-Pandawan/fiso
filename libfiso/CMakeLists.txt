set(FISO_SOURCES src/util.cpp src/layout.cpp src/image.cpp src/writer.cpp src/sources.cpp
                src/create.cpp src/ops.cpp src/capi.cpp
                src/cd_core.cpp src/cd_mmc.cpp src/cd_macos.cpp src/cd_linux.cpp src/cd_windows.cpp src/cd_none.cpp)
if(FISO_BUILD_SHARED)
  add_library(fiso SHARED ${FISO_SOURCES})
  target_compile_definitions(fiso PUBLIC FISO_SHARED PRIVATE FISO_BUILDING)
  set_target_properties(fiso PROPERTIES VERSION 1.1.0 SOVERSION 1 CXX_VISIBILITY_PRESET hidden)
else()
  add_library(fiso STATIC ${FISO_SOURCES})
endif()
target_include_directories(fiso PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(fiso PRIVATE fiso_flac ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(fiso PRIVATE /W4)
else()
  target_compile_options(fiso PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
install(TARGETS fiso)
install(FILES include/fiso/fiso.h DESTINATION include/fiso)
